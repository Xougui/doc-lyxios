export const staff = await fetch('https://api-lyxios.xouxou-hosting.fr/api/v1/staff', {
    headers: {
      'User-Agent': 'LyxiosDocs/1.0',
      'Accept': 'application/json'
    }
  })
  .then((res) => res.ok ? res.json() : Promise.reject(`Erreur HTTP ${res.status}`))
  .then((data) => {
    // Gestion plus souple de la réponse API et logs de débogage
    if (Array.isArray(data)) return data;
    if (data.status === 'success' && Array.isArray(data.staff)) return data.staff;
    
    console.warn('⚠️ API Staff : Format de réponse inattendu.', JSON.stringify(data, null, 2));
    return [];
  })
  .catch((e) => {
    console.error('Erreur lors de la récupération du staff:', e);
    return [];
  });

export const getRoleIndex = (memberId) => {
  const roleOrder = [
    "Propriétaire", 
    "Gérant", 
    "Développeur",
    "Gérant Blacklist",
    "Community Manager",
    "Hébergeur",
    "Designeur",
  ];
  const info = customInfo[memberId];
  if (!info || !info.roles || info.roles.length === 0) {
    return roleOrder.length; // Sans rôle, trié en dernier
  }
  
  // Trouve l'index du rôle le plus haut placé que le membre possède
  const indices = info.roles.map(role => {
    const index = roleOrder.indexOf(role);
    return index === -1 ? roleOrder.length : index;
  });
  return Math.min(...indices);
};

export const sortedStaff = [...staff].sort((a, b) => {
  // Priorité 1: xougui
  if (String(a.id) === '946098490654740580') return -1;
  if (String(b.id) === '946098490654740580') return 1;

  // Priorité 2: kadawatcha
  if (String(a.id) === '1178647820052467823') return -1;
  if (String(b.id) === '1178647820052467823') return 1;

  // Tri par rôle pour les autres
  const roleIndexA = getRoleIndex(String(a.id));
  const roleIndexB = getRoleIndex(String(b.id));
  
  if (roleIndexA !== roleIndexB) {
    return roleIndexA - roleIndexB;
  }

  // Tri par nom en cas d'égalité de rôle pour stabiliser l'ordre
  return (a.display_name || a.name).localeCompare(b.display_name || b.name);
});

export const staffGroups = (() => {
  const groups = [
    {
      title: "Direction",
      check: (roles) => roles.includes("Propriétaire") || roles.includes("Gérant"),
      members: []
    },
    {
      title: "Développement",
      check: (roles) => roles.includes("Développeur"),
      members: []
    },
    {
      title: "Équipe",
      check: () => true,
      members: []
    }
  ];

  sortedStaff.forEach(member => {
    const info = customInfo[String(member.id)] || {};
    const roles = info.roles || [];
    
    for (const group of groups) {
      if (group.check(roles)) {
        group.members.push(member);
        break;
      }
    }
  });
  
  return groups;
})();
